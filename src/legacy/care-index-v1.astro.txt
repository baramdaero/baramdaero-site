---
// 세척·유지관리 랜딩 — 가격·문구는 전부 src/content/site/pricing.json (하드코딩 금지).
// 빈 값이면 해당 섹션 미렌더 — 헤더+하단 CTA만으로도 페이지가 성립한다.
import Base from '../../layouts/Base.astro';
import { SITE_CONFIG } from '../../config.js';
import { loadPricing } from '../../lib/pricing.ts';
import { loadFaq } from '../../lib/faq.ts';

const pricing = loadPricing();
const { prices, conditions, process, symptoms, seasons } = pricing;

const headline = pricing.headline || '세척·유지관리';
const intro = pricing.intro;

// faq_ref=true → faq.json에서 '세척' 관련 항목만 이 페이지에 노출 + FAQPage 마크업
const careFaq = pricing.faqRef
  ? loadFaq().filter((f) => f.q.includes('세척') || f.a.includes('세척'))
  : [];

const siteUrl = Astro.site?.href.replace(/\/$/, '') ?? '';

// 증상 카드 아이콘 — currentColor 라인 SVG (DESIGN.md: 이모지 금지)
const ICON_PATHS: Record<string, string> = {
  smell: 'M7 17c0-2 1.5-2.5 1.5-4.5S7 10 7 8s1.5-2.5 1.5-4.5 M12.5 17c0-2 1.5-2.5 1.5-4.5s-1.5-2.5-1.5-4.5 1.5-2.5 1.5-4.5 M18 17c0-2 1.5-2.5 1.5-4.5S18 10 18 8s1.5-2.5 1.5-4.5',
  wind: 'M3 8h11a3 3 0 1 0-3-3 M3 13h15a3 3 0 1 1-3 3 M3 18h7',
  stain: 'M4 14h16l-2 4H6l-2-4Z M8 11v.01 M12 10v.01 M16 11v.01 M12 6v.01',
  drip: 'M12 4c3 4 5 6.5 5 9a5 5 0 0 1-10 0c0-2.5 2-5 5-9Z',
};

/* ---------- JSON-LD ---------- */
const serviceJsonLd = {
  '@context': 'https://schema.org',
  '@type': 'Service',
  serviceType: '에어컨 분해세척·유지관리',
  provider: { '@type': 'HVACBusiness', name: SITE_CONFIG.SITE_NAME, url: siteUrl },
  areaServed: { '@type': 'AdministrativeArea', name: '수도권' },
  ...(prices.length
    ? {
        offers: prices.map((p) => ({
          '@type': 'Offer',
          itemOffered: { '@type': 'Service', name: `${p.type} 분해세척` },
          ...(p.unit.includes('만원')
            ? {
                priceSpecification: {
                  '@type': 'UnitPriceSpecification',
                  minPrice: p.base * 10000,
                  priceCurrency: 'KRW',
                },
              }
            : {}),
        })),
      }
    : {}),
};

const breadcrumbJsonLd = {
  '@context': 'https://schema.org',
  '@type': 'BreadcrumbList',
  itemListElement: [
    { '@type': 'ListItem', position: 1, name: '홈', item: `${siteUrl}/` },
    { '@type': 'ListItem', position: 2, name: '세척·관리', item: `${siteUrl}/care/` },
  ],
};

const jsonLd: object[] = [serviceJsonLd, breadcrumbJsonLd];
if (careFaq.length) {
  jsonLd.push({
    '@context': 'https://schema.org',
    '@type': 'FAQPage',
    mainEntity: careFaq.map((f) => ({
      '@type': 'Question',
      name: f.q,
      acceptedAnswer: { '@type': 'Answer', text: f.a },
    })),
  });
}

<Base title={SITE_CONFIG.CARE_SEO_TITLE} description={SITE_CONFIG.CARE_SEO_DESCRIPTION} jsonLd={jsonLd}>
  <section class="bd-section care-hero">

              </button>

        </div>

    </section>
  )}

  {prices.length > 0 && (
    <section class="bd-section care-prices">

        <h2 class="bd-section-title">기준가</h2>

          </thead>
          <tbody>
            {prices.map((p) => (
              <tr>
                <th scope="row">{p.type}</th>
                <td class="care-prices__value">{p.base.toLocaleString('ko-KR')}<span class="care-prices__unit">{p.unit}</span></td>
                <td class="care-prices__note">{p.note}</td>
              </tr>
            ))}
          </tbody>
        </table>
        {conditions.length > 0 && (
          <ul class="care-prices__conditions">
            {conditions.map((c) => <li>{c}</li>)}
          </ul>
        )}
      </div>
    </section>
  )}

  {process.length > 0 && (
    <section class="bd-section care-process">
      <div class="bd-container">
        <p class="bd-microlabel">Process</p>
        <h2 class="bd-section-title">세척 절차</h2>
        <ol class="care-process__list">
          {process.map((st) => (
            <li class="care-process__item">
              <span class="care-process__no">{String(st.step).padStart(2, '0')}</span>
              <div class="care-process__body">
                <h3>{st.title}</h3>
                <p>{st.body}</p>
                {st.image && <img src={st.image} alt={`${st.title} 작업 사진`} loading="lazy" />}
              </div>
            </li>
          ))}
        </ol>
      </div>
    </section>
  )}

  {seasons.length > 0 && (
    <section class="bd-section care-seasons">
      <div class="bd-container">
        <p class="bd-microlabel">Seasons</p>
        <h2 class="bd-section-title">시기별 세척 이유</h2>
        <div class="care-seasons__grid">
          {seasons.map((s) => (
            <div class="care-seasons__card">
              <h3>{s.season}</h3>
              <p>{s.body}</p>
            </div>
          ))}
        </div>
      </div>
    </section>
  )}

  {careFaq.length > 0 && (
    <section class="bd-section care-faq">
      <div class="bd-container">
        <p class="bd-microlabel">FAQ</p>
        <h2 class="bd-section-title">세척 관련 질문</h2>
        <dl class="care-faq__list">
          {careFaq.map((f) => (
            <div class="care-faq__item">
              <dt>{f.q}</dt>
              <dd>{f.a}</dd>
            </div>
          ))}
        </dl>
      </div>
    </section>
  )}

  <section class="bd-section care-cta">
    <div class="bd-container care-cta__inner">
      <h2 class="care-cta__title">전면·측면 사진이면 충분합니다.</h2>
      <p class="care-cta__desc">기종과 대수를 선택하고 사진을 보내주시면 확정 견적을 드립니다.</p>
      <button type="button" class="bd-cta" data-chat-tree="clean">사진 보내고 확정 견적 받기</button>
    </div>
  </section>
</Base>
